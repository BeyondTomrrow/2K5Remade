cmake_minimum_required(VERSION 3.20)
project(GuestMemoryPosixCheck LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Guest memory POSIX checks require Linux")
endif()
add_executable(guest_memory_check check.c ../../external/xboxrecomp/src/platform/guest_memory.c)
target_include_directories(guest_memory_check PRIVATE ../../external/xboxrecomp/src/platform)
set_target_properties(guest_memory_check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_compile_options(guest_memory_check PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME guest_memory_boundary COMMAND guest_memory_check)
